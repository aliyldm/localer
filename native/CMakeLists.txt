add_library(localer_core STATIC room.cpp engine.cpp platform.cpp)
target_include_directories(localer_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(localer_core PUBLIC Qt6::Core Qt6::Network)
target_compile_definitions(localer_core PUBLIC LOCALER_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_link_libraries(localer_core PUBLIC advapi32 shell32)
endif()
add_library(localer_ui STATIC mainwindow.cpp resources.qrc)
target_link_libraries(localer_ui PUBLIC localer_core Qt6::Widgets)
qt_add_executable(LocalerQt WIN32 main.cpp)
target_link_libraries(LocalerQt PRIVATE localer_ui)
if(WIN32)
  enable_language(RC)
  target_sources(LocalerQt PRIVATE windows/app.rc)
  if(MSVC)
    target_link_options(LocalerQt PRIVATE /MANIFEST:NO)
  endif()
endif()
set_target_properties(LocalerQt PROPERTIES OUTPUT_NAME Localer)
add_executable(localer_tests tests/core_tests.cpp)
target_link_libraries(localer_tests PRIVATE localer_core Qt6::Test)
target_compile_definitions(localer_tests PRIVATE LOCALER_VECTOR_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol-vectors.json")
add_executable(localer_ui_tests tests/ui_tests.cpp)
target_link_libraries(localer_ui_tests PRIVATE localer_ui Qt6::Test)
add_executable(localer_network_tests tests/network_tests.cpp)
target_link_libraries(localer_network_tests PRIVATE localer_core Qt6::Test)
add_executable(localer_tun_tests tests/tun_tests.cpp)
target_link_libraries(localer_tun_tests PRIVATE localer_core Qt6::Test)
add_test(NAME core COMMAND localer_tests)
add_test(NAME ui COMMAND localer_ui_tests)
if(WIN32)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows")
else()
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
foreach(target localer_core localer_ui LocalerQt localer_tests localer_ui_tests localer_network_tests localer_tun_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
